/* ============================================================
   ESTADO GLOBAL (persistido en localStorage)
   ============================================================ */
const STORAGE_KEY = 'musk-algorithm-v1';

const defaultState = {
  currentStep: 0,
  requirements: [],   // {id, text, isPhysical, challenged}
  eliminations: [],   // {id, text, restored}
  simplifications: [],// {id, text, optimized}
  cycles: [],         // {id, text, timeBefore, timeAfter}
  automations: []     // {id, text, automated}
};

let state = loadState();

function loadState() {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    return raw ? { ...defaultState, ...JSON.parse(raw) } : { ...defaultState };
  } catch { return { ...defaultState }; }
}
function saveState() {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
}
function uid() { return Math.random().toString(36).slice(2, 9); }

/* ============================================================
   DATOS DE LOS PASOS
   ============================================================ */
const STEPS = [
  {
    id: 0,
    title: "Cuestiona cada requisito",
    subtitle: "Desafía toda suposición que no se base en leyes de la física",
    description: `Antes de optimizar cualquier cosa, pregúntate: <strong>¿esto realmente necesita existir?</strong>
    Un requisito solo es válido si se basa en una verdad fundamental (física, matemática, legal inmutable).
    Todo lo demás es sospechoso y debe ser cuestionado.`
  },
  {
    id: 1,
    title: "Elimina lo innecesario",
    subtitle: "Si no reinstauras ≥10% de lo eliminado, no eliminaste suficiente",
    description: `Elimina cualquier parte o proceso que no aporte valor esencial.
    <strong>Regla de Musk:</strong> si al final no has tenido que restaurar al menos el 10% de lo que quitaste,
    significa que no fuiste lo suficientemente agresivo.`
  },
  {
    id: 2,
    title: "Simplifica y optimiza",
    subtitle: "Solo después de eliminar, perfecciona lo que queda",
    description: `Optimizar algo que no debería existir es el error más común.
    Una vez que solo tienes lo esencial, hazlo más simple, más claro, más directo.`
  },
  {
    id: 3,
    title: "Acelera el tiempo de ciclo",
    subtitle: "Hazlo más rápido, pero solo si ya está optimizado",
    description: `Acelerar un proceso ineficiente solo produce errores más rápido.
    Una vez simplificado, reduce el tiempo entre inicio y fin.`
  },
  {
    id: 4,
    title: "Automatiza",
    subtitle: "Aplica tecnología al final, sobre un sistema ya eficiente",
    description: `Automatizar el caos genera caos automatizado.
    Solo cuando el proceso es simple, rápido y probado, aplícale tecnología.`
  }
];

/* ============================================================
   RENDERIZADO
   ============================================================ */
const app = document.getElementById('app');
const progressBar = document.getElementById('progressBar');

function renderProgress() {
  progressBar.innerHTML = STEPS.map((s, i) => {
    const cls = i === state.currentStep ? 'active' : (i < state.currentStep ? 'done' : '');
    return `<div class="step-dot ${cls}" onclick="goToStep(${i})" title="${s.title}">${i + 1}</div>`;
  }).join('');
}

function render() {
  renderProgress();
  const step = STEPS[state.currentStep];
  app.innerHTML = `
    <div class="card">
      <span class="step-number">Paso ${step.id + 1} de 5</span>
      <h2>${step.title}</h2>
      <p class="step-desc"><em>${step.subtitle}</em></p>
      <p class="step-desc">${step.description}</p>
      ${renderStepContent(step.id)}
      <div class="nav-buttons">
        <button class="secondary" onclick="goToStep(${state.currentStep - 1})" ${state.currentStep === 0 ? 'disabled style="opacity:0.3"' : ''}>← Anterior</button>
        ${state.currentStep < 4
          ? `<button onclick="goToStep(${state.currentStep + 1})">Siguiente →</button>`
          : `<button onclick="goToStep(5)">📋 Ver resumen</button>`}
      </div>
    </div>
  `;
}

/* ============================================================
   CONTENIDO POR PASO
   ============================================================ */
function renderStepContent(stepId) {
  if (stepId === 0) return renderStep0();
  if (stepId === 1) return renderStep1();
  if (stepId === 2) return renderStep2();
  if (stepId === 3) return renderStep3();
  if (stepId === 4) return renderStep4();
  if (stepId === 5) return renderSummary();
}

function renderStep0() {
  return `
    <label>Requisito o tarea a cuestionar</label>
    <input type="text" id="reqInput" placeholder="Ej: 'Necesitamos una reunión semanal de 1 hora'" onkeydown="if(event.key==='Enter')addRequirement()">
    <label>¿Se basa en una ley de la física? (física, matemática, legal inmutable)</label>
    <select id="reqPhysical" style="width:100%;background:var(--bg);border:1px solid var(--border);color:var(--text);padding:10px;border-radius:8px;">
      <option value="no">No — es una suposición / convención</option>
      <option value="yes">Sí — es una verdad fundamental</option>
    </select>
    <button onclick="addRequirement()">+ Añadir requisito</button>

    <ul class="item-list" id="reqList">
      ${state.requirements.map(r => `
        <li>
          <div class="content">
            <div>${escapeHtml(r.text)}</div>
            <div class="meta">${r.isPhysical ? '✅ Verdad fundamental' : '⚠️ Suposición — cuestionable'}</div>
          </div>
          <button class="small danger" onclick="removeRequirement('${r.id}')">✕</button>
        </li>
      `).join('') || '<li style="color:var(--muted);justify-content:center;">Aún no has añadido requisitos</li>'}
    </ul>
  `;
}

function renderStep1() {
  const total = state.eliminations.length;
  const restored = state.eliminations.filter(e => e.restored).length;
  const pct = total ? Math.round((restored / total) * 100) : 0;
  const enough = pct >= 10;
  return `
    <label>Elemento a eliminar (tarea, proceso, herramienta, reunión…)</label>
    <input type="text" id="elimInput" placeholder="Ej: 'Informe diario de 30 min'" onkeydown="if(event.key==='Enter')addElimination()">
    <button onclick="addElimination()">+ Añadir a la lista de eliminación</button>

    <div style="margin-top:20px;">
      <strong>Progreso de eliminación</strong>
      <span class="counter ${enough ? 'ok' : 'warn'}">${restored}/${total} reinstaurado (${pct}%)</span>
      <p style="font-size:0.8rem;color:var(--muted);margin-top:6px;">
        Regla: si no reinstauras ≥10%, no eliminaste lo suficiente. Marca lo que tuviste que recuperar.
      </p>
    </div>

    <ul class="item-list" id="elimList">
      ${state.eliminations.map(e => `
        <li class="${e.restored ? 'removed' : ''}">
          <div class="content">
            <div>${escapeHtml(e.text)}</div>
            <div class="meta">${e.restored ? '↩️ Reinstaurado (no era eliminable)' : '🗑️ Eliminado'}</div>
          </div>
          <button class="small secondary" onclick="toggleRestore('${e.id}')">${e.restored ? 'Re-eliminar' : 'Reinstaurar'}</button>
          <button class="small danger" onclick="removeElimination('${e.id}')">✕</button>
        </li>
      `).join('') || '<li style="color:var(--muted);justify-content:center;">Nada eliminado aún</li>'}
    </ul>

    <div class="warning-box ${total > 0 && !enough ? 'show' : ''}">
      ⚠️ Aún no has reinstaurado el 10%. Considera eliminar más agresivamente.
    </div>
  `;
}

function renderStep2() {
  return `
    <label>¿Qué queda y cómo lo simplificas?</label>
    <input type="text" id="simpInput" placeholder="Ej: 'Unificar 3 reportes en 1 dashboard'" onkeydown="if(event.key==='Enter')addSimplification()">
    <button onclick="addSimplification()">+ Añadir simplificación</button>

    <ul class="item-list">
      ${state.simplifications.map(s => `
        <li>
          <div class="content">
            <div>${escapeHtml(s.text)}</div>
            <div class="meta">✂️ Simplificado</div>
          </div>
          <button class="small danger" onclick="removeItem('simplifications','${s.id}')">✕</button>
        </li>
      `).join('') || '<li style="color:var(--muted);justify-content:center;">Aún sin simplificaciones</li>'}
    </ul>
  `;
}

function renderStep3() {
  return `
    <label>Proceso a acelerar</label>
    <input type="text" id="cycleInput" placeholder="Ej: 'Aprobación de gastos'" onkeydown="if(event.key==='Enter')addCycle()">
    <label>Tiempo actual (ej: "3 días")</label>
    <input type="text" id="cycleBefore" placeholder="3 días">
    <label>Tiempo objetivo (ej: "4 horas")</label>
    <input type="text" id="cycleAfter" placeholder="4 horas">
    <button onclick="addCycle()">+ Añadir ciclo a acelerar</button>

    <ul class="item-list">
      ${state.cycles.map(c => `
        <li>
          <div class="content">
            <div>${escapeHtml(c.text)}</div>
            <div class="meta">⏱️ ${escapeHtml(c.timeBefore)} → ${escapeHtml(c.timeAfter)}</div>
          </div>
          <button class="small danger" onclick="removeItem('cycles','${c.id}')">✕</button>
        </li>
      `).join('') || '<li style="color:var(--muted);justify-content:center;">Sin ciclos definidos</li>'}
    </ul>
  `;
}

function renderStep4() {
  return `
    <label>Proceso a automatizar (ya debe estar simplificado y rápido)</label>
    <input type="text" id="autoInput" placeholder="Ej: 'Envío de facturas recurrentes'" onkeydown="if(event.key==='Enter')addAutomation()">
    <label>Herramienta / tecnología</label>
    <input type="text" id="autoTool" placeholder="Ej: 'Zapier + Stripe'">
    <button onclick="addAutomation()">+ Añadir automatización</button>

    <ul class="item-list">
      ${state.automations.map(a => `
        <li>
          <div class="content">

          </div>

        </li>
      `).join('') || '<li style="color:var(--muted);justify-content:center;">Sin automatizaciones</li>'}
    </ul>
  `;
}

function renderSummary() {
  const s = state;
  return `
    <div class="summary">
      <h3 style="margin-bottom:16px;">📋 Tu plan de simplificación</h3>
      <pre>${buildSummaryText()}</pre>
      <button onclick="copySummary()">📋 Copiar resumen</button>
      <button class="secondary" onclick="resetAll()">🗑️ Reiniciar todo</button>
    </div>
  `;
}

/* ============================================================
   LÓGICA DE DATOS
   ============================================================ */
function addRequirement() {
  const input = document.getElementById('reqInput');
  const physical = document.getElementById('reqPhysical').value === 'yes';
  const text = input.value.trim();
  if (!text) return;
  state.requirements.push({ id: uid(), text, isPhysical: physical });
  input.value = '';
  saveState(); render();
}

function removeRequirement(id) {
  state.requirements = state.requirements.filter(r => r.id !== id);
  saveState(); render();
}

function addElimination() {
  const input = document.getElementById('elimInput');
  const text = input.value.trim();
  if (!text) return;
  state.eliminations.push({ id: uid(), text, restored: false });
  input.value = '';
  saveState(); render();
}

function toggleRestore(id) {
  const e = state.eliminations.find(x => x.id === id);
  if (e) e.restored = !e.restored;
  saveState(); render();
}

function removeElimination(id) {
  state.eliminations = state.eliminations.filter(e => e.id !== id);
  saveState(); render();
}

function addSimplification() {
  const input = document.getElementById('simpInput');
  const text = input.value.trim();
  if (!text) return;
  state.simplifications.push({ id: uid(), text });
  input.value = '';
  saveState(); render();
}

function addCycle() {
  const text = document.getElementById('cycleInput').value.trim();
  const before = document.getElementById('cycleBefore').value.trim();
  const after = document.getElementById('cycleAfter').value.trim();
  if (!text) return;
  state.cycles.push({ id: uid(), text, timeBefore: before || '?', timeAfter: after || '?' });
  saveState(); render();
}

function addAutomation() {
  const text = document.getElementById('autoInput').value.trim();
  const tool = document.getElementById('autoTool').value.trim();
  if (!text) return;
  state.automations.push({ id: uid(), text, tool: tool || 'sin definir' });
  saveState(); render();
}

function removeItem(key, id) {
  state[key] = state[key].filter(x => x.id !== id);
  saveState(); render();
}

function goToStep(i) {
  if (i < 0 || i > 5) return;
  state.currentStep = i;
  saveState(); render();
  window.scrollTo({ top: 0, behavior: 'smooth' });
}

function resetAll() {
  if (!confirm('¿Seguro que quieres borrar todo tu progreso?')) return;
  state = { ...defaultState };
  saveState(); render();
}

/* ============================================================
   UTILIDADES
   ============================================================ */
function escapeHtml(str) {
  const div = document.createElement('div');
  div.textContent = str;
  return div.innerHTML;
}

function buildSummaryText() {
  const s = state;
  let out = '=== ALGORITMO DE MUSK — RESUMEN ===\n\n';

  out += '1. REQUISITOS CUESTIONADOS\n';
  if (s.requirements.length === 0) out += '   (ninguno)\n';
  s.requirements.forEach(r => {
    out += `   ${r.isPhysical ? '✅' : '⚠️'} ${r.text}\n`;
  });

  out += '\n2. ELIMINACIONES\n';
  if (s.eliminations.length === 0) out += '   (ninguna)\n';
  s.eliminations.forEach(e => {
    out += `   ${e.restored ? '↩️ Reinstaurado:' : '🗑️ Eliminado:'} ${e.text}\n`;
  });
  const restored = s.eliminations.filter(e => e.restored).length;
  const pct = s.eliminations.length ? Math.round((restored / s.eliminations.length) * 100) : 0;
  out += `   → ${pct}% reinstaurado ${pct >= 10 ? '(✔ suficiente)' : '(⚠️ elimina más)'}\n`;

  out += '\n3. SIMPLIFICACIONES\n';
  if (s.simplifications.length === 0) out += '   (ninguna)\n';
  s.simplifications.forEach(x => out += `   ✂️ ${x.text}\n`);

  out += '\n4. CICLOS ACELERADOS\n';
  if (s.cycles.length === 0) out += '   (ninguno)\n';
  s.cycles.forEach(c => out += `   ⏱️ ${c.text}: ${c.timeBefore} → ${c.timeAfter}\n`);

  out += '\n5. AUTOMATIZACIONES\n';
  if (s.automations.length === 0) out += '   (ninguna)\n';
  s.automations.forEach(a => out += `   🤖 ${a.text} [${a.tool}]\n`);

  out += '\n=== FIN ===';
  return out;
}

function copySummary() {
  const text = buildSummaryText();
  navigator.clipboard.writeText(text).then(() => {
    alert('✅ Resumen copiado al portapapeles');
  }).catch(() => {
    alert('No se pudo copiar. Selecciona el texto manualmente.');
  });
}

render();
